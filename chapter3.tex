\chapter{Ranks}
\section{Ramanujan's and Dyson's work}
Ramanujan's classical congruences state that, for every $k\ge0$,
\begin{align}
 p(5k+4)&\equiv0\pmod5,\label{eq:ramanujan5}\\
 p(7k+5)&\equiv0\pmod7,\label{eq:ramanujan7}\\
 p(11k+6)&\equiv0\pmod{11}.\label{eq:ramanujan11}
\end{align}
We give a proof of \eqref{eq:ramanujan5}; see Berndt~\cite{berndt} for
further background on these congruences.
\begin{proof}[Proof of \eqref{eq:ramanujan5}]
Put $E=(q;q)_\infty$ and $J=E^3$. Euler's pentagonal number theorem and
Jacobi's identity for the cube of the Euler product give
\[
 E=\sum_{n\in\mathbb Z}(-1)^nq^{n(3n+1)/2},\qquad
 J=\sum_{n\ge0}(-1)^n(2n+1)q^{n(n+1)/2}.
\]
The exponents in $E$ are congruent to $0$, $1$, or $2$ modulo $5$.
Those in $J$ are congruent to $0$, $1$, or $3$; in the last case
$n\equiv2\pmod5$, so $5\mid2n+1$.
If $E_i$ and $J_i$ denote the corresponding residue-class subseries, then
\[
 E=E_0+E_1+E_2,\qquad J\equiv J_0+J_1\pmod5.
\]
Furthermore, the binomial theorem gives
$E^5\equiv(q^5;q^5)_\infty\pmod5$. Thus
\[
 \sum_{n\ge0}p(n)q^n=\frac{EJ}{E^5}
 \equiv\frac{(E_0+E_1+E_2)(J_0+J_1)}{(q^5;q^5)_\infty}\pmod5.
\]
The numerator has no terms whose exponents are $4$ modulo $5$, and the
reciprocal denominator contains only multiples of $5$. Comparing
coefficients proves the claim.
\end{proof}

The first two congruences also follow immediately from the identities
\begin{align}
 \sum_{k\ge0}p(5k+4)q^k
 &=5\frac{(q^5;q^5)_\infty^5}{(q;q)_\infty^6},\label{eq:p5-dissection}\\
 \sum_{k\ge0}p(7k+5)q^k
 &=7\frac{(q^7;q^7)_\infty^3}{(q;q)_\infty^4}
   +49q\frac{(q^7;q^7)_\infty^7}{(q;q)_\infty^8}.
 \label{eq:p7-dissection}
\end{align}
These are recorded here as known identities~\cite{berndt,berndt2}; the
proof above does not by itself establish the congruences modulo $7$ or $11$.

\subsection{The rank of a partition}
Dyson~\cite{dyson} defined the rank of a nonempty partition by
\[
 r(\lambda)=l(\lambda)-n(\lambda).
\]
For its conjugate partition $\lambda^*$, we have
$r(\lambda^*)=-r(\lambda)$. In particular, the rank distribution is symmetric.
The empty partition is assigned rank zero.

Write $N(s,n)$ for the number of partitions of $n$ with rank $s$, and
$N(s,m,n)$ for the number whose rank is $s$ modulo $m$. Then
\[
 N(s,m,n)=\sum_{k\in\mathbb Z}N(mk+s,n).
\]
Dyson conjectured the equalities
\[
 N(s,5,5n+4)=\frac{p(5n+4)}5,\qquad
 N(t,7,7n+5)=\frac{p(7n+5)}7
\]
for every residue $s$ modulo $5$, every residue $t$ modulo $7$, and $n\ge0$.
These equalities divide the relevant partitions into equally sized classes,
giving combinatorial interpretations of Ramanujan's first two congruences.

Atkin and Swinnerton-Dyer~\cite{atkin1954} proved Dyson's conjectures by
finding generating functions for rank differences
\[
 N(s,\ell,\ell n+d)-N(t,\ell,\ell n+d)
\]
with $\ell=5,7$ and $0\le s,t,d<\ell$. The differences vanish for
$(\ell,d)=(5,4)$ and $(7,5)$.

\subsection{The \texorpdfstring{$M_2$}{M2}-rank}
For a partition $\lambda$ without repeated odd parts, define
\[
 r_2(\lambda)=\ceil*{\frac{l(\lambda)}2}-n(\lambda).
\]
Even parts may be repeated. Again, the empty partition has rank zero.
Let $N_2(s,n)$ count these partitions with $M_2$-rank equal to $s$, and
let $N_2(s,m,n)$ count those with rank congruent to $s$ modulo $m$.
Differences between these counts are the subject of the next chapter.
