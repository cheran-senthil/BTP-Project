\chapter{Fundamental properties of \texorpdfstring{$q$}{q}-series and theta functions}
\label{ch:qseries}
\section{The \texorpdfstring{$q$}{q}-binomial theorem}
The $q$-binomial theorem generalizes the binomial expansion: replacing $a$
by $q^\alpha$ and taking $q\to1^-$ recovers the series for $(1-z)^{-\alpha}$.
\begin{theorem}\label{thm:qbinomial}
For $|q|<1$ and $|z|<1$,
\begin{equation}\label{eq:qbinomial}
 \sum_{n\ge0}\frac{(a;q)_n}{(q;q)_n}z^n
 =\frac{(az;q)_\infty}{(z;q)_\infty}.
\end{equation}
\end{theorem}
\begin{proof}
The product on the right converges uniformly on compact subsets of
$|z|<1$ and defines an analytic function there. Write
\[
 F(z)=\frac{(az;q)_\infty}{(z;q)_\infty}=\sum_{n\ge0}A_nz^n.
\]
The product representation gives $(1-z)F(z)=(1-az)F(qz)$.
Comparing coefficients for $n\ge1$ yields
\[
 A_n-A_{n-1}=q^nA_n-aq^{n-1}A_{n-1},\qquad
 A_n=\frac{1-aq^{n-1}}{1-q^n}A_{n-1}.
\]
Since $A_0=1$, induction gives $A_n=(a;q)_n/(q;q)_n$.
\end{proof}

\subsection{Euler's identities}
Two useful consequences are
\begin{align}
 \sum_{n\ge0}\frac{z^n}{(q;q)_n}
 &=\frac1{(z;q)_\infty}\qquad(|z|<1),\label{eq:euler-reciprocal}\\
 \sum_{n\ge0}\frac{(-z)^nq^{n(n-1)/2}}{(q;q)_n}
 &=(z;q)_\infty.\label{eq:euler-product}
\end{align}
The first follows by setting $a=0$ in \eqref{eq:qbinomial}.
For the second, replace $z$ by $z/a$ and let $a\to\infty$;
the resulting identity holds for all complex $z$ when $|q|<1$.

\section{The Jacobi triple product identity}
\begin{theorem}\label{thm:jacobi}
For $z\ne0$ and $0<|q|<1$,
\begin{equation}\label{eq:jacobi}
 \sum_{n\in\mathbb Z}z^nq^{n^2}
 =(-zq;q^2)_\infty(-q/z;q^2)_\infty(q^2;q^2)_\infty.
\end{equation}
\end{theorem}
\begin{proof}
Applying \eqref{eq:euler-product} with base $q^2$ gives
\begin{align*}
 (-zq;q^2)_\infty
 &=\sum_{n\ge0}\frac{z^nq^{n^2}}{(q^2;q^2)_n}\\
 &=\frac1{(q^2;q^2)_\infty}
   \sum_{n\in\mathbb Z}z^nq^{n^2}(q^{2n+2};q^2)_\infty.
\end{align*}
The extension to negative $n$ is valid because the product in each such
term contains the factor $1-q^0=0$.
Expand that product using \eqref{eq:euler-product}. Initially assuming
$|q/z|<1$, absolute convergence permits rearrangement, giving
\begin{align*}
 (-zq;q^2)_\infty
 &=\frac1{(q^2;q^2)_\infty}
   \sum_{n\in\mathbb Z}\sum_{r\ge0}
   \frac{z^n(-1)^rq^{n^2+(2n+2)r+r(r-1)}}{(q^2;q^2)_r}\\
 &=\frac1{(q^2;q^2)_\infty}
   \sum_{r\ge0}\frac{(-q/z)^r}{(q^2;q^2)_r}
   \sum_{m\in\mathbb Z}z^mq^{m^2}\\
 &=\frac{\sum_{m\in\mathbb Z}z^mq^{m^2}}
   {(q^2;q^2)_\infty(-q/z;q^2)_\infty}.
\end{align*}
In the second line we set $m=n+r$; the third line uses
\eqref{eq:euler-reciprocal}. Rearranging proves \eqref{eq:jacobi} for
$|z|>|q|$. Both sides of \eqref{eq:jacobi} are analytic for $z\ne0$,
so analytic continuation proves the identity throughout that domain.
\end{proof}

\subsection{Euler's pentagonal number theorem}
A specialization of \eqref{eq:jacobi} gives
\begin{equation}\label{eq:pentagonal}
 (q;q)_\infty
 =\sum_{n\in\mathbb Z}(-1)^nq^{n(3n-1)/2}
 =\sum_{n\in\mathbb Z}(-1)^nq^{n(3n+1)/2}.
\end{equation}
The two sums agree after replacing $n$ by $-n$. These identities are also
valid as formal power series.
