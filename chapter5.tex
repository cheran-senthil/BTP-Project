\chapter{Cubic partition-pair congruences}
\section{Definitions}
\subsection{Cubic partitions and overcubic partitions}
Chan~\cite{chan1,chan2,chan3} studied the cubic partition function $a(n)$,
defined by
\[
 \sum_{n\ge0}a(n)q^n
 =\frac1{(q;q)_\infty(q^2;q^2)_\infty}.
\]
A cubic partition may be viewed as an ordered pair of partitions, the
second containing only even parts. Kim~\cite{kim1} introduced the
corresponding overcubic partition function
\[
 \sum_{n\ge0}\overline a(n)q^n
 =\frac{(-q;q)_\infty(-q^2;q^2)_\infty}
 {(q;q)_\infty(q^2;q^2)_\infty}.
\]

\subsection{Cubic partition pairs}
Zhao and Zhong~\cite{zhao1} studied the function $b(n)$ defined by
\begin{equation}\label{eq:b-gf}
 \sum_{n\ge0}b(n)q^n
 =\frac1{(q;q)_\infty^2(q^2;q^2)_\infty^2}.
\end{equation}
It counts ordered pairs of cubic partitions with total size $n$.
Kim~\cite{kim2} studied statistics explaining congruences such as
$b(5n+4)\equiv0\pmod5$.
The overpartition analogue~\cite{kim3} is
\[
 \sum_{n\ge0}\overline b(n)q^n
 =\frac{(-q;q)_\infty^2(-q^2;q^2)_\infty^2}
 {(q;q)_\infty^2(q^2;q^2)_\infty^2}.
\]

\section{Preliminaries}
Throughout this part, write
\[
 f_t=(q^t;q^t)_\infty\quad(t\ge1),\qquad
 \mathcal A(q)=\sum_{m,n\in\mathbb Z}q^{m^2+mn+n^2}.
\]
The symbol $\mathcal A(q)$ denotes a theta series, distinct from the
generating function for $a(n)$. Euler's pentagonal number theorem gives
\[
 f_1=\sum_{m\in\mathbb Z}(-1)^mq^{m(3m+1)/2}.
\]

\begin{lemma}\label{lem:product-congruences}
For every positive integer $t$,
\[
 f_t^3\equiv f_{3t}\pmod3,\qquad
 f_t^9\equiv f_{3t}^3\pmod9.
\]
\end{lemma}
\begin{proof}
For each factor, $(1-x)^3\equiv1-x^3\pmod3$. Writing
$(1-x)^3=1-x^3+3R(x)$ and cubing shows
$(1-x)^9\equiv(1-x^3)^3\pmod9$. Multiplication over the factors proves
the assertions coefficientwise.
\end{proof}
All $f_t$ have constant term one and are therefore invertible as formal
power series. The congruences may consequently also be used in quotients.

The product evaluations of the theta functions yield
\[
 \varphi(-q)=\frac{f_1^2}{f_2},\qquad
 \psi(q)=\frac{f_2^2}{f_1}.
\]
We also use the following dissection identities~\cite{zhao1}:
\begin{align}
 \frac1{f_1f_2}
 &=\mathcal A(q^6)\frac{f_9^3}{f_3^4f_6^3}
   +q\mathcal A(q^3)\frac{f_{18}^3}{f_3^3f_6^4}
   +3q^2\frac{f_9^3f_{18}^3}{f_3^4f_6^4},\label{eq:cubic-reciprocal}\\
 f_1f_2&=\frac{f_6f_9^4}{f_3f_{18}^2}
   -qf_9f_{18}-2q^2\frac{f_3f_{18}^4}{f_6f_9^2}.
 \label{eq:cubic-product}
\end{align}
Each product quotient multiplying $1$, $q$, or $q^2$ is a series in $q^3$.
This makes extraction of a specified residue class possible.

\section{Lin's congruence modulo 27}
Lin~\cite{Lin2017} established the generating-function congruence
\begin{equation}\label{eq:lin27-gf}
 \sum_{n\ge0}b(9n+7)q^n
 \equiv18f_1f_2\left(\frac{f_3^7}{f_6}
      +q\frac{f_6^7}{f_3}\right)-9f_3^2f_6^2\pmod{27}.
\end{equation}
We use it to deduce the following result, following Lin's argument.
\begin{theorem}[Lin]\label{thm:lin27}
For every $n\ge0$, $b(81n+61)\equiv0\pmod{27}$.
\end{theorem}
\begin{proof}
Substitute \eqref{eq:cubic-product} into \eqref{eq:lin27-gf}, retain the
terms whose exponents are divisible by $3$, and replace $q^3$ by $q$.
This gives
\begin{align}
 \sum_{n\ge0}b(27n+7)q^n
 &\equiv18\left(\frac{f_1^6f_3^4}{f_6^2}
     -2q\frac{f_2^6f_6^4}{f_3^2}\right)-9f_1^2f_2^2
       \pmod{27}\notag\\
 &\equiv18\left(\frac{f_3^6}{f_6^2}
     -2q\frac{f_6^6}{f_3^2}\right)
     -9\frac{f_3f_6}{f_1f_2}\pmod{27}.
 \label{eq:b27-mod27}
\end{align}
In the second line we used Lemma~\ref{lem:product-congruences} modulo $3$;
the prefactors $18$ and $9$ make this sufficient modulo $27$.
Substituting \eqref{eq:cubic-reciprocal} into the last term yields
\begin{equation*}
\begin{split}
 \sum_{n\ge0}b(27n+7)q^n
 &\equiv18\left(\frac{f_3^6}{f_6^2}-2q\frac{f_6^6}{f_3^2}\right)\\
 &\quad-9f_3f_6\left(
 \mathcal A(q^6)\frac{f_9^3}{f_3^4f_6^3}
 +q\mathcal A(q^3)\frac{f_{18}^3}{f_3^3f_6^4}
 \right)\pmod{27}.
\end{split}
\end{equation*}
The omitted $q^2$ term has a coefficient divisible by $27$. Every
remaining term has exponent congruent to $0$ or $1$ modulo $3$.
The coefficient of $q^{3n+2}$ on the left is
$b(27(3n+2)+7)=b(81n+61)$, which proves the theorem.
\end{proof}
