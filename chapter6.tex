\chapter{Resolutions of Lin's conjectures}\label{ch:lin}
\section{Lin's conjectures and subsequent proofs}
Lin~\cite{Lin2017} conjectured the stronger congruence
\begin{equation}\label{eq:lin243}
 b(81n+61)\equiv0\pmod{243}\qquad(n\ge0)
\end{equation}
and the generating-function congruences
\begin{align}
 \sum_{n\ge0}b(81n+7)q^n
 &\equiv9\frac{f_2f_3^2}{f_6}\pmod{81},\label{eq:lin81a}\\
 \sum_{n\ge0}b(81n+34)q^n
 &\equiv36\frac{f_1f_6^2}{f_3}\pmod{81}.\label{eq:lin81b}
\end{align}
These are now theorems. Chern~\cite{chern} resolved the conjectures and
proved the stronger divisibility $b(81n+61)\equiv0\pmod{729}$.
Gireesh and Mahadeva Naika~\cite{gireesh} subsequently gave a general family
of congruences and alternative proofs of Lin's conjectures. This chapter
presents the deductions used in the original project report.

\section{The general family of congruences}
Gireesh and Mahadeva Naika use identities of Cooper~\cite{cooper} together
with the huffing operator
\[
 H\left(\sum_{n\ge0}c_nq^n\right)=\sum_{n\ge0}c_{3n}q^{3n}.
\]
Their recurrence and valuation arguments establish the following theorem.
\begin{theorem}[Gireesh--Mahadeva Naika, \cite{gireesh}]
\label{thm:general-cubic}
For all integers $\alpha,n\ge0$,
\begin{align}
 b\left(3^{2\alpha+1}n+\frac{3^{2\alpha+1}+1}{4}\right)
 &\equiv0\pmod{3^{2\alpha}},\label{eq:family1}\\
 b\left(3^{2\alpha+4}n+\frac{3^{2\alpha+5}+1}{4}\right)
 &\equiv0\pmod{3^{2\alpha+5}},\label{eq:family2}\\
 b\left(3^{2\alpha+5}n+\frac{7\cdot3^{2\alpha+4}+1}{4}\right)
 &\equiv0\pmod{3^{\alpha+6}},\label{eq:family3}\\
 b\left(3^{2\alpha+5}n+\frac{11\cdot3^{2\alpha+4}+1}{4}\right)
 &\equiv0\pmod{3^{\alpha+6}}.\label{eq:family4}
\end{align}
\end{theorem}
Setting $\alpha=0$ in \eqref{eq:family2} gives \eqref{eq:lin243} immediately.
We cite the general theorem rather than reproduce its recurrence argument.

\section{The two congruences modulo 81}
We give the deduction in~\cite[Section 5]{gireesh}, making explicit the
product identities and coefficient extractions it uses.
\begin{lemma}\label{lem:five-power-dissection}
\begin{equation}\label{eq:five-power-dissection}
\begin{split}
 \frac{f_2^5}{f_1^4}
 &=\frac{f_6^6f_9^{10}}{f_3^{10}f_{18}^5}
   +4q\frac{f_6^5f_9^7}{f_3^9f_{18}^2}
   +9q^2\frac{f_6^4f_9^4f_{18}}{f_3^8}\\
 &\quad+10q^3\frac{f_6^3f_9f_{18}^4}{f_3^7}
   +4q^4\frac{f_6^2f_{18}^7}{f_3^6f_9^2}.
\end{split}
\end{equation}
\end{lemma}
\begin{proof}
We use the two identities recorded in~\cite[Section 5]{gireesh}:
\begin{align}
 \frac{f_2^2}{f_1}
 &=\frac{f_6f_9^2}{f_3f_{18}}+q\frac{f_{18}^2}{f_9},
 \label{eq:psi-dissection}\\
 \frac{f_2}{f_1^2}
 &=\frac{f_6^4f_9^6}{f_3^8f_{18}^3}
   +2q\frac{f_6^3f_9^3}{f_3^7}
   +4q^2\frac{f_6^2f_{18}^3}{f_3^6}.
 \label{eq:hirschhorn-dissection}
\end{align}
The first is a corollary in Berndt~\cite[p.~49]{berndt2}; the second is
used by Hirschhorn and Sellers~\cite{sellers}. Squaring
\eqref{eq:psi-dissection} and multiplying by
\eqref{eq:hirschhorn-dissection} gives \eqref{eq:five-power-dissection}.
\end{proof}

The next two inputs are the congruences recorded in~\cite[Section 5]{gireesh}:
\begin{align}
 U(q):=\frac{f_2^4f_3^8}{f_1^8f_6^4}
       +q\frac{f_2f_6^5}{f_1^5f_3}
 &\equiv1\pmod9,\label{eq:U-congruence}\\
 \sum_{n\ge0}b(27n+7)q^n
 &\equiv9\frac{f_3^2f_6^2}{f_1^4f_2^4}
 \equiv9\frac{f_3^2f_2^5}{f_1^4f_6}\pmod{81}.
 \label{eq:b27-mod81}
\end{align}
The first follows from a theta identity of Son~\cite{hwan}.
The first congruence in \eqref{eq:b27-mod81} is an input from the
recurrence calculation in~\cite{gireesh}; the second uses
$f_2^9\equiv f_6^3\pmod9$.

\begin{theorem}
The congruences \eqref{eq:lin81a} and \eqref{eq:lin81b} hold.
\end{theorem}
\begin{proof}
Combining \eqref{eq:five-power-dissection} and \eqref{eq:b27-mod81} gives
\begin{equation}\label{eq:b27-expanded}
\begin{split}
 \sum_{n\ge0}b(27n+7)q^n
 &\equiv9\frac{f_3^2}{f_6}\left(
 \frac{f_6^6f_9^{10}}{f_3^{10}f_{18}^5}
 +4q\frac{f_6^5f_9^7}{f_3^9f_{18}^2}\right.\\
 &\hspace{25mm}\left.
 +q^3\frac{f_6^3f_9f_{18}^4}{f_3^7}
 +4q^4\frac{f_6^2f_{18}^7}{f_3^6f_9^2}
 \right)\pmod{81}.
\end{split}
\end{equation}
The coefficient $9q^2$ in the dissection vanishes after multiplication
by $9$ modulo $81$, and $10$ may be replaced by $1$ in the $q^3$ term.

Retain the terms with exponents divisible by $3$ in
\eqref{eq:b27-expanded} and replace $q^3$ by $q$. We obtain
\begin{align*}
 \sum_{n\ge0}b(81n+7)q^n
 &\equiv9\frac{f_1^2}{f_2}\left(
 \frac{f_2^6f_3^{10}}{f_1^{10}f_6^5}
 +q\frac{f_2^3f_3f_6^4}{f_1^7}\right)\pmod{81}\\
 &\equiv9\frac{f_2f_3^2}{f_6}U(q)\pmod{81}.
\end{align*}
Using \eqref{eq:U-congruence} proves \eqref{eq:lin81a}.

Next retain the terms with exponents congruent to $1$ modulo $3$, divide
by $q$, and replace $q^3$ by $q$. Then
\begin{align*}
 \sum_{n\ge0}b(81n+34)q^n
 &\equiv36\frac{f_1^2}{f_2}\left(
 \frac{f_2^5f_3^7}{f_1^9f_6^2}
 +q\frac{f_2^2f_6^7}{f_1^6f_3^2}\right)\pmod{81}\\
 &\equiv36\frac{f_1f_6^2}{f_3}U(q)\pmod{81}.
\end{align*}
Since $36\cdot9$ is divisible by $81$, \eqref{eq:U-congruence} proves
\eqref{eq:lin81b} as well.
\end{proof}
