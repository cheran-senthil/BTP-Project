\chapter{Introduction to partitions}
\section{Background}
A partition of a nonnegative integer $n$ is a nonincreasing sequence of
positive integers whose sum is $n$. Two sums differing only in the order
of their summands represent the same partition. The notation
$\lambda\vdash n$ means that $\lambda$ is a partition of $n$. The empty
partition is the unique partition of zero.

A summand is called a part. For a nonempty partition $\lambda$, write
$n(\lambda)$ for its number of parts and $l(\lambda)$ for its largest part.
For example, the five partitions of $4$ are
\[
 (4),\quad(3,1),\quad(2,2),\quad(2,1,1),\quad(1,1,1,1).
\]
Thus $n((2,1,1))=3$ and $l((2,1,1))=2$.

\section{Definitions and terminology}
\subsection{The \texorpdfstring{$q$}{q}-Pochhammer symbol}
The $q$-Pochhammer symbol is defined by
\[
 (a;q)_0=1,\qquad (a;q)_n=\prod_{k=0}^{n-1}(1-aq^k)\quad(n\ge1),
\]
and
\[
 (a;q)_\infty=\prod_{k=0}^\infty(1-aq^k)\qquad(|q|<1).
\]
When the base is clear, we abbreviate $(a;q)_n$ to $(a)_n$.
The following product identities will be useful:
\begin{align}
 (q^a;q^b)_\infty(-q^a;q^b)_\infty
 &= (q^{2a};q^{2b})_\infty,\label{eq:product-sign}\\
 (cq^a;q^{2b})_\infty(cq^{a+b};q^{2b})_\infty
 &= (cq^a;q^b)_\infty.\label{eq:product-split}
\end{align}
Here the parameters are chosen so that the products converge. For the
partition-generating products below, the same identities hold formally.
We also use the notation
\begin{align*}
 (a_1,\ldots,a_k;q)_n&=\prod_{i=1}^k(a_i;q)_n,\\
 J_b&=(q^b;q^b)_\infty,\\
 J_{a,b}&=(q^a,q^{b-a},q^b;q^b)_\infty\quad(0<a<b),\\
 \chi(q)&=(-q;q^2)_\infty.
\end{align*}

\subsection{Ramanujan's general theta function}
Ramanujan's general theta function is
\begin{equation}
 f(a,b)=\sum_{n\in\mathbb Z}a^{n(n+1)/2}b^{n(n-1)/2},\qquad |ab|<1.
\end{equation}
The specializations used in this report are
\begin{align*}
 \varphi(q)&=f(q,q)=\sum_{n\in\mathbb Z}q^{n^2}
             =(-q,-q,q^2;q^2)_\infty,\\
 \psi(q)&=f(q,q^3)=\sum_{n\ge0}q^{n(n+1)/2}
             =\frac{(q^2;q^2)_\infty}{(q;q^2)_\infty}.
\end{align*}
These product evaluations follow from the Jacobi triple product in
Chapter~\ref{ch:qseries}; see also Berndt~\cite{berndt,berndt2}.

\subsection{Partition functions}
Let $p(n)$ denote the number of partitions of $n$, with $p(0)=1$.
Write $p_m(n)$ for the number of partitions whose parts are at most $m$.
For a set $S$ of positive integers, let $p(S,m,n)$ count partitions of $n$
into exactly $m$ parts from $S$. When $S$ is the set of all positive
integers, abbreviate this to $p(m,n)$.

Similarly, $Q(n)$ counts partitions into distinct parts, and $Q(S,m,n)$
counts partitions of $n$ into exactly $m$ distinct parts from $S$.
We use $Q(m,n)$ when $S$ is unrestricted, and set $Q(0)=1$.

\subsection{Overpartitions}
An overpartition is a partition in which the first occurrence of each
distinct part may be overlined. The eight overpartitions of $3$ are
\[
 (3),\ (\overline3),\ (2,1),\ (2,\overline1),\
 (\overline2,1),\ (\overline2,\overline1),\
 (1,1,1),\ (\overline1,1,1).
\]
Writing $\overline p(n)$ for their number, we have
\[
 \overline p(n)=\sum_{k=0}^n p(n-k)Q(k).
\]
Indeed, the overlined parts form a partition into distinct parts, while
the non-overlined parts form an unrestricted partition.

\section{Generating functions}
The generating function of a sequence $(a_n)_{n\ge0}$ is the formal power
series $\sum_{n\ge0}a_nq^n$. For ordinary partitions, choosing the
multiplicity of each part independently gives
\begin{equation}\label{eq:partition-gf}
 \sum_{n\ge0}p(n)q^n
 =\prod_{k\ge1}(1+q^k+q^{2k}+\cdots)
 =\frac1{(q;q)_\infty}.
\end{equation}
More generally,
\begin{align*}
 \sum_{n,m\ge0}p(S,m,n)z^mq^n&=\prod_{k\in S}\frac1{1-zq^k},\\
 \sum_{n,m\ge0}Q(S,m,n)z^mq^n&=\prod_{k\in S}(1+zq^k).
\end{align*}
The convolution for overpartitions therefore gives
\[
 \sum_{n\ge0}\overline p(n)q^n
 =\frac{(-q;q)_\infty}{(q;q)_\infty}.
\]
All coefficient extractions in this report are formal. Analytic identities
are initially stated in their domains of convergence. Congruences between
series with integer coefficients are understood coefficientwise.
